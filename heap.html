using namespace std;

void heapify(vector>& h, int& n) {
    int i, j, k;
    pair v;
    bool heap;
    for (i = n / 2; i >= 1; i--) {
        k = i;
        v = h[k];
        heap = false;
        while (!heap && 2 * k <= n) {
            j = 2 * k;
            if (j < n && h[j].second < h[j + 1].second)
                j = j + 1;
            if (v.second >= h[j].second)
                heap = true;
            else {
                h[k] = h[j];
                k = j;
            }
        }
        h[k] = v;
    }
}

void heapsort(vector>& h, int& n, vector>& d) {
    while (h.size() != 0) {
        d.push_back(h[1]);
        swap(h[1], h[n]);
        h.pop_back();
        n--;
        heapify(h, n);
    }
}

void heapsort_7() {
    vector> h(1);
    int n = 5, l = 5;

    h.push_back({1, 10});
    h.push_back({2, 20});
    h.push_back({3, 15});
    h.push_back({4, 30});
    h.push_back({5, 40});

    heapify(h, n);

    cout << "\n\nConsidering the input as the 'safety bar' : higher the value, higher the risk\n";
    cout << "\nZones before prioritization\n";

    for (int i = 1; i <= l; i++) {
        cout << "Location : " << h[i].first << " , Risk " << h[i].second << "\n";
    }
    cout << "\n\nZones after prioritization\n";

    vector> d(1);
    heapsort(h, n, d);

    for (int i = 1; i <= l; i++) {
        cout << "Location : " << d[i].first << " , Risk " << d[i].second << "\n";
    }
}
